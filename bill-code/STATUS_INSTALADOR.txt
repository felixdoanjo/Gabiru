  🔧 COMANDOS ÚTEIS

# Compilar TypeScript
npm run electron:compile

# Testar app
npm run electron:dev

# Limpar runtime (simular 1ª execução)
Remove-Item "$env:APPDATA\bill-code\runtime" -Recurse -Force

# Ver tamanho do runtime
Get-ChildItem "$env:APPDATA\bill-code\runtime" -Recurse | 
  Measure-Object -Property Length -Sum | 
  Select-Object @{Name="Size(GB)";Expression={[math]::Round($_.Sum/1GB,2)}}

# Gerar instalador
npm run build

# Ver tamanho do instalador
Get-Item "release\Gabiru-Setup*.exe" | 
  Select-Object Name, @{Name="Size(MB)";Expression={[math]::Round($_.Length/1MB,2)}}
